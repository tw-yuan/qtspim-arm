# Test manifest for scripts/run-tests.sh.
#
# One test per line:  name | stdin | spim arguments
#
#   name   Test name; the expected output is in tests/golden/<name>.stdout,
#          <name>.stderr and <name>.status (exit status).
#   stdin  File fed to spim's standard input, or "-" for /dev/null.
#   args   Arguments passed to spim, split on white space (no quoting).
#
# spim runs in the repository root with an empty environment (env -i), so
# all paths are relative to the repository root.  The exception handler is
# always given explicitly (-ef CPU/exceptions.s) or disabled, so the result
# does not depend on the default handler path compiled into spim.

# Upstream torture tests, with the arguments used by spim/Makefile (test and
# test_bare targets).

tt.le          | -                      | -ef CPU/exceptions.s -file Tests/tt.le.s

tt.alu.bare    | -                      | -bare -noexception -file Tests/tt.alu.bare.s
tt.fpu.bare    | -                      | -bare -noexception -file Tests/tt.fpu.bare.s
